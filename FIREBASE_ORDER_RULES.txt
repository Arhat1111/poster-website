INKWAVES FIRESTORE RULES — V8.5

Paste the rules below into Firebase Console -> Firestore -> Rules and click Publish.
These rules keep products/categories public-readable and admin-editable, while allowing customers to CREATE checkout orders without being able to read or edit other orders.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null
        && request.auth.uid == "WV1xGunguGhnMEl5XmelNztNVZx2";
    }

    match /products/{productId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /categories/{categoryId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /settings/{settingId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /orders/{orderId} {
      allow create: if
        request.resource.data.paymentStatus == 'pending'
        && request.resource.data.fulfillmentStatus == 'New'
        && request.resource.data.customer is map
        && request.resource.data.customer.email is string
        && request.resource.data.customer.phone is string
        && request.resource.data.items is list
        && request.resource.data.items.size() > 0
        && request.resource.data.items.size() <= 30
        && request.resource.data.totals is map
        && request.resource.data.totals.total is number
        && request.resource.data.totals.total > 0
        && request.resource.data.totals.total < 100000;

      allow read, update, delete: if isAdmin();
    }

    match /orders/{orderId}/photos/{photoId} {
      allow create: if
        request.resource.data.data is string
        && request.resource.data.data.size() < 900000
        && request.resource.data.data.matches('^data:image/.*');
      allow read, update, delete: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
